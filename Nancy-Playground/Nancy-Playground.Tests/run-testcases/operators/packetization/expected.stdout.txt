// Packetization, the use of floor and ceil that network calculus makes
// The two curves named below are how the operators were spelled before they existed:
// they are kept here to check that the operators compute the same thing
> oldFloor := right-ext(stair(1, 1, 1))
oldFloor = upp(period([(0, 0)] ](0, 0) 0 (1, 0)[), 1, 1)
> oldCeil := stair(0, 1, 1)
oldCeil = stair(0, 1, 1)
> f := bucket(3, 2)
f = bucket(3, 2)
> L := 5
L = 5
// only whole packets of size L leave a packetizer
> packetized := floor(f / L) * L
packetized = upp(period([(0, 0)] ](0, 0) 0 (1, 0)[ [(1, 5)] ](1, 5) 0 (5/3, 5)[), 5, 5/3)
> assert(packetized = (oldFloor comp (f / L)) * L)
true
// a packetizer outputs no more than its input, and lags by less than one packet
> assert(packetized <= f)
true
> assert(f <= packetized + L)
true
// the number of packets needed to carry f, i.e. the arrival curve a per-packet resource sees
> packets := ceil(f / L)
packets = upp([(0, 0)] ](0, 1) 0 (1, 1)[, period([(1, 1)] ](1, 2) 0 (8/3, 2)[), 1, 5/3)
> assert(packets = oldCeil comp (f / L))
true
> assert(packets >= f / L)
true
// a periodic task with period T and workload C per release
> T := 4
T = 4
> C := 3
C = 3
> release := ceil(affine(1 / T, 0)) * C
release = upp(period([(0, 0)] ](0, 3) 0 (4, 3)[), 3, 4)
> assert(release = stair(0, T, C))
true
> assert(release = (oldCeil comp affine(1 / T, 0)) * C)
true
> release(9)
9
