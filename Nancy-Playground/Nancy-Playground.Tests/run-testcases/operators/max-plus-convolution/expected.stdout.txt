// Test (max,+) convolution: f1 *^ f2
// f1(x) = 2x + 5
// f2(x) = 3x + 2
// Expected result: (f1 *^ f2)(t) = 3t + 7 (slope 2\/3=3, burst 5+2=7)
> f1 := affine(2, 5)
f1 = uaf([(0, 5)] ](0, 5) 2 (+inf, +inf)[)
> f2 := affine(3, 2)
f2 = uaf([(0, 2)] ](0, 2) 3 (+inf, +inf)[)
> result := f1 *^ f2
result = uaf([(0, 7)] ](0, 7) 3 (+inf, +inf)[)
> expected := affine(3, 7)
expected = uaf([(0, 7)] ](0, 7) 3 (+inf, +inf)[)
> assert(result = expected)
true
