// Test floor and ceil, which return the kind of their argument
// On a curve they round its values, on a scalar they round the value itself
f := affine(1/3, 0)
floorF := floor(f)
ceilF := ceil(f)
assert( floorF(5) = 1 )
assert( ceilF(5) = 2 )
// the floor of a curve is what the composition with right-ext(stair(1, 1, 1)) used to compute
old := right-ext(stair(1, 1, 1)) comp f
assert( floorF = old )
// on scalars, where the result stays a scalar
x := 7/2
floor(x)
ceil(x)
floor(-x)
ceil(-x)
// dividing one floor by another is not integer division
floor(x) / floor(9/2)
// a scalar floor multiplying a curve scales it
scaled := floor(x) * f
assert( scaled(3) = 3 )
