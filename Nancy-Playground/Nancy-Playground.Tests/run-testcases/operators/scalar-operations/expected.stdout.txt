// The scalar operators: abs, pow, mod, gcd and lcm
> abs(-7/2)
7/2
> abs(-inf)
+inf
> pow(2, 10)
1024
> pow(2, -2)
1/4
> pow(-7/2, 3)
-343/8
// mod is infix, like div, and the remainder takes the sign of the dividend
> 7 mod 3
1
> -7 mod 3
-1
// gcd and lcm are defined on rationals, not only on integers
> gcd(12, 18)
6
> lcm(4, 6)
12
> gcd(1/2, 1/3)
1/6
> lcm(1/2, 1/3)
1
// they compose with each other, and with the operators of the same version
> x := 4
x = 4
> y := 6
y = 6
> assert(gcd(x, y) * lcm(x, y) = x * y)
true
> assert(abs(-7 mod 3) = 1)
true
> assert(floor(pow(3, 2) / 2) = 4)
true
// and are a scalar wherever a scalar goes
> f := affine(1, 0)
f = uaf([(0, 0)] ](0, 0) 1 (+inf, +inf)[)
> scaled := abs(-3) * f
scaled = uaf([(0, 0)] ](0, 0) 3 (+inf, +inf)[)
> assert(scaled(2) = 6)
true
> shifted := f + (7 mod 3)
shifted = uaf([(0, 1)] ](0, 1) 1 (+inf, +inf)[)
> shifted(0)
1
