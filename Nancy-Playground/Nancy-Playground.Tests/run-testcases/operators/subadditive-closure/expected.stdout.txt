// Test subadditive closure: star(f)
// star(f) = lim_{n->inf} f^{*n} / n
// For affine(r, 0), star returns the same function since it's already subadditive
> f := affine(3, 0)
f = uaf([(0, 0)] ](0, 0) 3 (+inf, +inf)[)
> star_f := star(f)
star_f = uaf([(0, 0)] ](0, 0) 3 (+inf, +inf)[)
> assert(f = star_f)
true
> g := ratency(1, 1) + 1
g = uaf([(0, 1)] ](0, 1) 0 (1, 1)[ [(1, 1)] ](1, 1) 1 (+inf, +inf)[)
> star_g := star(g)
star_g = uaf([(0, 0)] ](0, 1) 0 (1, 1)[ [(1, 1)] ](1, 1) 1 (+inf, +inf)[)
> assert(g >= star_g)
true
